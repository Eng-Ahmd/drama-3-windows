cmake_minimum_required(VERSION 3.20)
project(drama3_cuda_bridge LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Windows MSVC required")
endif()
get_filename_component(DRAMA3_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(DONOR_BUILD "${DRAMA3_ROOT}/builds/s2-cuda" CACHE PATH "Pinned CUDA donor")
set(CPU_DONOR_BUILD "${DONOR_BUILD}" CACHE PATH "Pinned CPU backend donor (same GGML revision)")
foreach(runtime ggml-base ggml-cpu ggml ggml-cuda)
    add_library(${runtime} SHARED IMPORTED)
    set(runtime_donor "${DONOR_BUILD}")
    if(runtime STREQUAL "ggml-cpu")
        set(runtime_donor "${CPU_DONOR_BUILD}")
    endif()
    if(runtime STREQUAL "ggml-cuda")
        set(implib "${runtime_donor}/ggml/src/ggml-cuda/ggml-cuda.lib")
    else()
        set(implib "${runtime_donor}/ggml/src/${runtime}.lib")
    endif()
    set_target_properties(${runtime} PROPERTIES IMPORTED_IMPLIB "${implib}"
        IMPORTED_LOCATION "${runtime_donor}/bin/${runtime}.dll")
endforeach()
add_library(drama3-experts-bridge SHARED ../experts_bridge.cpp)
target_compile_features(drama3-experts-bridge PRIVATE cxx_std_17)
target_compile_options(drama3-experts-bridge PRIVATE /W4 /WX /EHsc)
target_compile_definitions(drama3-experts-bridge PRIVATE DRAMA3_CUDA)
target_include_directories(drama3-experts-bridge PRIVATE "${DRAMA3_ROOT}/upstream/s2.cpp/ggml/include")
target_link_libraries(drama3-experts-bridge PRIVATE ggml-base ggml-cpu ggml ggml-cuda)
foreach(runtime ggml-base ggml-cpu ggml ggml-cuda)
    add_custom_command(TARGET drama3-experts-bridge POST_BUILD COMMAND ${CMAKE_COMMAND} -E
        copy_if_different "$<TARGET_FILE:${runtime}>" "$<TARGET_FILE_DIR:drama3-experts-bridge>")
endforeach()
